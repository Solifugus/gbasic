# G-SQL's parser corpus (tests/gsql/sql_parse_test.c). `>` is the SQL, `=` the
# canonical form it must print as, `! CODE POSITION text` a refusal: the code,
# the 1-based character position, and a part of the message.

# ---- SELECT ---------------------------------------------------------
> select 1
= SELECT 1
> SELECT a, b FROM t
= SELECT a, b FROM t
> select * from Orders;
= SELECT * FROM orders
> select "Mixed", "select" from "T"
= SELECT "Mixed", "select" FROM "T"
> select o.* from orders o
= SELECT o.* FROM orders AS o
> select a as x, b y from t as u
= SELECT a AS x, b AS y FROM t AS u
> select a + b * c - d from t
= SELECT ((a + (b * c)) - d) FROM t
> select (a + b) * c from t
= SELECT ((a + b) * c) FROM t
> select -a, - -2, +3 from t
= SELECT (- a), (- (- 2)), 3 FROM t
> select a || b || 'x' from t
= SELECT ((a || b) || 'x') FROM t
> select a from t where a = 1 and b <> 2 or not c != 3
= SELECT a FROM t WHERE (((a = 1) AND (b <> 2)) OR (NOT (c <> 3)))
> select a from t where a < 1 or a <= 2 or a > 3 or a >= 4
= SELECT a FROM t WHERE ((((a < 1) OR (a <= 2)) OR (a > 3)) OR (a >= 4))
> select a from t where a is null and b is not null and c is unknown
= SELECT a FROM t WHERE (((a IS NULL) AND (b IS NOT NULL)) AND (c IS NULL))
> select a from t where a in (1, 2, null) and b not in ('x')
= SELECT a FROM t WHERE ((a IN (1, 2, NULL)) AND (b NOT IN ('x')))
> select a from t where a between 1 and 10 and b not between :lo and $2
= SELECT a FROM t WHERE ((a BETWEEN 1 AND 10) AND (b NOT BETWEEN :lo AND $2))
> select a from t where name like 'A%' and name not like '_b'
= SELECT a FROM t WHERE ((name LIKE 'A%') AND (name NOT LIKE '_b'))
> select case when a > 1 then 'big' when a = 1 then 'one' else 'small' end from t
= SELECT CASE WHEN (a > 1) THEN 'big' WHEN (a = 1) THEN 'one' ELSE 'small' END FROM t
> select case a when 1 then 'one' end from t
= SELECT CASE a WHEN 1 THEN 'one' END FROM t
> select cast(a as money(usd)), cast('3' as integer), date '2026-10-30', timestamp '2026-10-30 12:00' from t
= SELECT CAST(a AS MONEY(USD)), CAST('3' AS INTEGER), CAST('2026-10-30' AS DATE), CAST('2026-10-30 12:00' AS DATETIME) FROM t
> select coalesce(a, b, 0), lower(name) from t
= SELECT coalesce(a, b, 0), lower(name) FROM t
> select p.vendor, p.address.city from assets
= SELECT p.vendor, p.address.city FROM assets
> select a.end from t a
= SELECT a."end" FROM t AS a
> select 'it''s', '' from t
= SELECT 'it''s', '' FROM t
> select 1.5, 2e3, .25, 9223372036854775807, 9223372036854775808
= SELECT 1.5, 2000.0, 0.25, 9223372036854775807, 9.2233720368547758e+18
> select true, false, null, unknown
= SELECT TRUE, FALSE, NULL, NULL
> select a from t order by a desc, b asc, 2 limit 10 offset :skip
= SELECT a FROM t ORDER BY a DESC, b, 2 LIMIT 10 OFFSET :skip
> select a from t offset 5 rows fetch first 3 rows only
= SELECT a FROM t LIMIT 3 OFFSET 5
> select a from t fetch next row only
= SELECT a FROM t LIMIT 1
> select a -- a comment
>   from /* another */ t
= SELECT a FROM t
> explain select a from t where id = :id
= EXPLAIN SELECT a FROM t WHERE (id = :id)
> explain delete from t where a = 1
= EXPLAIN DELETE FROM t WHERE (a = 1)

# ---- writes ---------------------------------------------------------
> insert into t values (1, 'a'), (2, default)
= INSERT INTO t VALUES (1, 'a'), (2, DEFAULT)
> insert into t (b, a) values (:b, $1)
= INSERT INTO t (b, a) VALUES (:b, $1)
> update t set a = a + 1, b = default where id = 3
= UPDATE t SET a = (a + 1), b = DEFAULT WHERE (id = 3)
> update t as x set a = 0
= UPDATE t AS x SET a = 0
> delete from t
= DELETE FROM t
> delete from t where a is null
= DELETE FROM t WHERE (a IS NULL)

# ---- schema ---------------------------------------------------------
> create table orders (id integer primary key, customer text not null, total money(usd) default cast('0' as money(usd)), placed date, notes record, paid boolean null)
= CREATE TABLE orders (id INTEGER PRIMARY KEY, customer TEXT NOT NULL, total MONEY(USD) DEFAULT CAST('0' AS MONEY(USD)), placed DATE, notes RECORD, paid BOOLEAN)
> create table if not exists t (a int, b bigint, c int64, d varchar, e string, f numeric, g real, h double precision, i timestamp, j datetime, k time, l bool, m money, primary key (a, b), unique (c), unique (d, e))
= CREATE TABLE IF NOT EXISTS t (a INTEGER, b INTEGER, c INTEGER, d TEXT, e TEXT, f NUMBER, g NUMBER, h NUMBER, i DATETIME, j DATETIME, k TIME, l BOOLEAN, m MONEY, PRIMARY KEY (a, b), UNIQUE (c), UNIQUE (d, e))
> create table t (email text unique not null)
= CREATE TABLE t (email TEXT NOT NULL UNIQUE)
> drop table t
= DROP TABLE t
> drop table if exists t;
= DROP TABLE IF EXISTS t
> alter table t add column note text default 'none'
= ALTER TABLE t ADD COLUMN note TEXT DEFAULT 'none'
> alter table t add n integer not null default 0
= ALTER TABLE t ADD COLUMN n INTEGER NOT NULL DEFAULT 0
> create index by_cust on orders (customer, placed desc)
= CREATE INDEX by_cust ON orders (customer, placed DESC)
> create unique index if not exists u on t (a asc)
= CREATE UNIQUE INDEX IF NOT EXISTS u ON t (a)
> drop index if exists by_cust
= DROP INDEX IF EXISTS by_cust

# ---- aggregates, GROUP BY, HAVING (docs/gsql_aggregate_design.md) -----
> select count(*) from t
= SELECT count(*) FROM t
> select a, count(distinct b), sum(c) from t group by a having count(*) > 1 order by 2 desc
= SELECT a, count(DISTINCT b), sum(c) FROM t GROUP BY a HAVING (count(*) > 1) ORDER BY 2 DESC
> select lower(b) from t where a > 0 group by lower(b), a
= SELECT lower(b) FROM t WHERE (a > 0) GROUP BY lower(b), a
> select max(a) from t having max(a) > 0
= SELECT max(a) FROM t HAVING (max(a) > 0)
> select sum(*) from t
! SYNTAX 12 only count takes *
> select a from t group a
! SYNTAX 23 expected BY

# ---- joins (docs/gsql_join_design.md) -------------------------------
> select a from t join u on t.x = u.x
= SELECT a FROM t JOIN u ON (t.x = u.x)
> select a from t inner join u on t.x = u.x left outer join v w on w.y = u.y
= SELECT a FROM t JOIN u ON (t.x = u.x) LEFT JOIN v AS w ON (w.y = u.y)
> select a from t, u cross join v
= SELECT a FROM t CROSS JOIN u CROSS JOIN v
> select e.n, m.n from emp e join emp m on e.boss = m.id where m.n like 'a%'
= SELECT e.n, m.n FROM emp AS e JOIN emp AS m ON (e.boss = m.id) WHERE (m.n LIKE 'a%')
> select a from t right join u on true
! UNSUPPORTED 17 RIGHT JOIN is not in this release
> select a from t full join u on true
! UNSUPPORTED 17 FULL JOIN is not in this release
> select a from t natural join u
! UNSUPPORTED 17 NATURAL JOIN is not in this release
> select a from t join u using (x)
! UNSUPPORTED 24 USING is not in this release
> select a from t join u
! SYNTAX 23 expected ON

# ---- SELECT DISTINCT (docs/gsql_distinct_design.md) ------------------
> select distinct a, b from t order by 1
= SELECT DISTINCT a, b FROM t ORDER BY 1
> select all a from t
= SELECT a FROM t
> select distinct * from t
= SELECT DISTINCT * FROM t
> select distinct on (a) a from t
! UNSUPPORTED 17 DISTINCT ON is not in this release
> select a, count(distinct b) from t group by a
= SELECT a, count(DISTINCT b) FROM t GROUP BY a

# ---- UNION, INTERSECT, EXCEPT (docs/gsql_setop_design.md) -------------
> select a from t union select b from u
= SELECT a FROM t UNION SELECT b FROM u
> select a from t union all select b from u intersect select c from v order by 1 limit 2
= SELECT a FROM t UNION ALL (SELECT b FROM u INTERSECT SELECT c FROM v) ORDER BY 1 LIMIT 2
> select a from t except distinct select b from u except all select c from v
= SELECT a FROM t EXCEPT SELECT b FROM u EXCEPT ALL SELECT c FROM v
> select a from t order by a union select b from u
! SYNTAX 28 ORDER BY, LIMIT and OFFSET go after the last SELECT
> select a from t union (select b from u)
= SELECT a FROM t UNION SELECT b FROM u
> select a from t intersect (select b from u union select c from v)
= SELECT a FROM t INTERSECT (SELECT b FROM u UNION SELECT c FROM v)
> select a from t union (select b from u limit 1)
! UNSUPPORTED 46 ORDER BY, LIMIT and OFFSET inside parentheses are not in this release
> select a from t union
! SYNTAX 22 expected SELECT
